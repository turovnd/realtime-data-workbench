"""Focused contract checks for the proposed Python style sample."""

import asyncio
import json
from collections.abc import Callable

import pytest
from pydantic import ValidationError

from enrich_feedback import (
    DEMO_FEEDBACK,
    Generate,
    enrich_feedback,
    generate_demo_result,
)


def response_adapter(payload: object) -> Generate:
    async def generate(_text: str) -> str:
        return json.dumps(payload)

    return generate


def valid_payload() -> dict[str, object]:
    return {
        "topic": "usability",
        "summary": "Live updates reset the selected filters.",
        "evidence": ["loses my filters"],
    }


def test_accepts_a_valid_synthetic_result() -> None:
    result = asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=generate_demo_result))

    assert result.topic == "usability"
    assert result.evidence == ("loses my filters",)


@pytest.mark.parametrize("evidence", [["keeps my filters"], ["   "]])
def test_rejects_missing_or_blank_evidence(evidence: list[str]) -> None:
    payload = valid_payload()
    payload["evidence"] = evidence

    with pytest.raises(ValueError, match="Evidence must quote"):
        asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=response_adapter(payload)))


@pytest.mark.parametrize(
    "mutate",
    [
        lambda payload: payload.update(topic="unknown"),
        lambda payload: payload.update(summary=123),
        lambda payload: payload.update(confidence=0.99),
        lambda payload: payload.update(evidence=[]),
    ],
)
def test_rejects_invalid_response_fields(
    mutate: Callable[[dict[str, object]], None],
) -> None:
    payload = valid_payload()
    mutate(payload)

    with pytest.raises(ValidationError):
        asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=response_adapter(payload)))


@pytest.mark.parametrize("text", ["", "   ", "x" * 4_001])
def test_validates_input_before_calling_the_adapter(text: str) -> None:
    async def unexpected_generate(_text: str) -> str:
        pytest.fail("The adapter must not receive invalid input.")

    with pytest.raises(ValidationError):
        asyncio.run(enrich_feedback(text, generate=unexpected_generate))


def test_rejects_invalid_json() -> None:
    async def generate(_text: str) -> str:
        return '{"topic":'

    with pytest.raises(ValidationError):
        asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=generate))


def test_rejects_oversized_output_before_parsing() -> None:
    async def generate(_text: str) -> str:
        return " " * 8_193

    with pytest.raises(ValueError, match="character limit"):
        asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=generate))


def test_preserves_an_adapter_failure() -> None:
    async def generate(_text: str) -> str:
        raise TimeoutError("Synthetic provider timeout")

    with pytest.raises(TimeoutError, match="Synthetic provider timeout"):
        asyncio.run(enrich_feedback(DEMO_FEEDBACK, generate=generate))
